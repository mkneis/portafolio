// Amortiguación crítica sin oscilación residual
export function solveSpring(
  mass: number,
  stiffness: number,
  damping: number
) {
  const w0 = Math.sqrt(stiffness / mass);
  const zeta = damping / (2 * Math.sqrt(stiffness * mass));

  return (t: number) => {
    const decay = Math.exp(-zeta * w0 * t);
    return 1 - decay * Math.cos(w0 * Math.sqrt(1 - zeta * zeta) * t);
  };
}
